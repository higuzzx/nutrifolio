function MetricBar({ label, value, current, color }: { label: string; value: string; current: string; color: string }) {
  const pct = label === 'Proteínas' ? '74%' : label === 'Carboidratos' ? '60%' : '64%';
  return (
    <View style={styles.metricRow}>
      <View style={styles.metricHeading}>
        <Text style={styles.metricLabel}>{label}</Text>
        <Text style={styles.metricValue}>{current} / {value}</Text>
      </View>
      <View style={styles.track}>
        <View style={[styles.trackFill, { width: pct, backgroundColor: color }]} />
      </View>
    </View>
  );
}

function AddMealModal({ visible, onClose }: { visible: boolean; onClose: () => void }) {
  const [query, setQuery] = useState('');
  const foods = useMemo(
    () =>
      [
        { name: 'Peito de frango grelhado', info: '100 g · 165 kcal' },
        { name: 'Arroz branco cozido', info: '100 g · 130 kcal' },
        { name: 'Ovo inteiro cozido', info: '1 unid. · 70 kcal' },
        { name: 'Banana prata', info: '1 unid. · 90 kcal' },
      ].filter((food) => food.name.toLowerCase().includes(query.toLowerCase())),
    [query]
  );
  return (
    <Modal animationType="slide" transparent visible={visible} onRequestClose={onClose}>
      <View style={styles.modalBackdrop}>
        <View style={styles.modalSheet}>
          <View style={styles.modalHandle} />
          <View style={styles.modalHeader}>
            <View>
              <Text style={styles.modalTitle}>Adicionar alimento</Text>
              <Text style={styles.modalSubtitle}>Monte sua refeição de forma simples.</Text>
            </View>
            <Pressable onPress={onClose} style={styles.closeButton}><X size={19} color={palette.text} /></Pressable>
          </View>
          <View style={styles.modalTabs}>
            <Text style={styles.modalTabActive}>Buscar alimentos</Text>
            <Text style={styles.modalTab}>Salvos</Text>
          </View>
          <View style={styles.searchBox}>
            <Search size={17} color={palette.muted} />
            <TextInput
              value={query}
              onChangeText={setQuery}
              placeholder="Buscar alimento"
              placeholderTextColor="#77736D"
              style={styles.searchInput}
            />
          </View>
          <Text style={styles.modalSectionLabel}>Sugestões populares</Text>
          <ScrollView style={styles.foodList} showsVerticalScrollIndicator={false}>
            {foods.map((food) => (
              <Pressable key={food.name} style={styles.foodRow} onPress={onClose}>
                <Image source={{ uri: chickenImage }} style={styles.foodImage} />
                <View style={styles.foodText}>
                  <Text style={styles.foodName}>{food.name}</Text>
                  <Text style={styles.foodInfo}>{food.info}</Text>
                </View>
                <View style={styles.foodAdd}><Plus size={16} color={palette.orange} /></View>
              </Pressable>
            ))}
          </ScrollView>
          <View style={styles.scannerPrompt}>
            <View style={styles.scannerIcon}><QrCode size={19} color={palette.orange} /></View>
            <View style={{ flex: 1 }}>
              <Text style={styles.scannerTitle}>Use o scanner</Text>
              <Text style={styles.scannerText}>Leia o código de barras de um produto.</Text>
            </View>
            <ChevronRight size={17} color={palette.muted} />
          </View>
          <Pressable style={styles.modalPrimaryButton} onPress={onClose}>
            <Text style={styles.modalPrimaryText}>Concluir refeição</Text>
          </Pressable>
        </View>
      </View>
    </Modal>
  );
}

export default function App() {
  const [activeTab, setActiveTab] = useState<Tab>('Início');
  const [modalVisible, setModalVisible] = useState(false);
  const { width } = useWindowDimensions();
  const contentWidth = Math.min(width, 520);

  const screen =
    activeTab === 'Início' ? <HomeScreen onAddMeal={() => setModalVisible(true)} />
    : activeTab === 'Diário' ? <DiaryScreen onAddMeal={() => setModalVisible(true)} />
    : activeTab === 'Progresso' ? <ProgressScreen />
    : <ProfileScreen />;

  const tabs: { label: Tab; icon: LucideIcon }[] = [
    { label: 'Início', icon: HomeIcon },
    { label: 'Diário', icon: BookOpen },
    { label: 'Progresso', icon: BarChart3 },
    { label: 'Perfil', icon: UserRound },
  ];

  return (
    <SafeAreaView style={styles.safeArea}>
      <View style={[styles.appShell, { width: contentWidth }]}>
        <View style={styles.appHeader}>
          <Pressable style={styles.headerIcon}><Menu size={21} color={palette.text} strokeWidth={1.8} /></Pressable>
          <Brand />
          <Pressable style={styles.headerIcon}><Camera size={19} color={palette.text} strokeWidth={1.8} /></Pressable>
        </View>
        <View style={styles.screenContent}>{screen}</View>
        <View style={styles.bottomNav}>
          {tabs.map(({ label, icon: Icon }) => (
            <Pressable key={label} style={styles.navItem} onPress={() => setActiveTab(label)}>
              <Icon size={20} color={activeTab === label ? palette.orange : '#77736D'} strokeWidth={activeTab === label ? 2.3 : 1.7} />
              <Text style={[styles.navLabel, activeTab === label && styles.navLabelActive]}>{label}</Text>
            </Pressable>
          ))}
          <Pressable style={styles.floatingAdd} onPress={() => setModalVisible(true)}>
            <Plus size={23} color="#0B0B0B" strokeWidth={2.8} />
          </Pressable>
        </View>
      </View>
      <AddMealModal visible={modalVisible} onClose={() => setModalVisible(false)} />
    </SafeAreaView>
  );
}
